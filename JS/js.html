📦 Variables et Déclarations

let, const et var

// ✅ let - variable réassignable avec portée de bloc let nom = 'Pierre'; nom = 'Marie'; // ✅ OK // ✅ const - constante, non réassignable const PI = 3.14159; // PI = 3.14; ❌ ERREUR // ⚠️ var - ancienne syntaxe (éviter) var age = 25;
💡 Bonne pratique : Utilisez const par défaut. N'utilisez let que si vous devez réassigner la variable. Évitez var.

Portée des variables

// Portée globale const global = 'accessible partout'; function exemple() { // Portée de fonction const locale = 'accessible dans la fonction'; if (true) { // Portée de bloc const bloc = 'accessible dans le bloc'; } // console.log(bloc); ❌ ERREUR }

Destructuring

// Destructuring d'objets const personne = { nom: 'Alice', age: 30, ville: 'Paris' }; const { nom, age } = personne; console.log(nom); // 'Alice' // Destructuring de tableaux const nombres = [1, 2, 3, 4]; const [premier, deuxieme, ...reste] = nombres; console.log(premier); // 1 console.log(reste); // [3, 4]

🔀 Structures Conditionnelles

if / else if / else

const note = 15; if (note >= 16) { console.log('Très bien'); } else if (note >= 14) { console.log('Bien'); } else if (note >= 12) { console.log('Assez bien'); } else { console.log('Peut mieux faire'); }

switch

const jour = 'lundi'; switch (jour) { case 'lundi': console.log('Début de semaine'); break; case 'vendredi': console.log('Bientôt le weekend!'); break; case 'samedi': case 'dimanche': console.log('Weekend!'); break; default: console.log('Jour de semaine'); }
💡 Astuce : N'oubliez pas le break dans chaque case du switch.

🔄 Boucles

Boucle for

// Boucle for classique for (let i = 0; i < 5; i++) { console.log(i); // 0, 1, 2, 3, 4 } // for...of (itération sur les valeurs) const fruits = ['pomme', 'banane', 'orange']; for (const fruit of fruits) { console.log(fruit); } // for...in (itération sur les clés) const personne = { nom: 'Alice', age: 30 }; for (const cle in personne) { console.log(cle, personne[cle]); }

Boucle while

let i = 0; while (i < 5) { console.log(i); i++; } // do...while (exécuté au moins une fois) let j = 0; do { console.log(j); j++; } while (j < 5);

break et continue

// break - sort de la boucle for (let i = 0; i < 10; i++) { if (i === 5) break; console.log(i); // 0, 1, 2, 3, 4 } // continue - passe à l'itération suivante for (let i = 0; i < 5; i++) { if (i === 2) continue; console.log(i); // 0, 1, 3, 4 }

🎯 Fonctions

Déclaration de Fonctions

// Fonction classique function saluer(nom) { return `Bonjour ${nom}`; } // Expression de fonction const additionner = function(a, b) { return a + b; }; // Fonction fléchée (ES6) ⭐ const multiplier = (a, b) => a * b; // Fonction fléchée avec bloc const calculer = (a, b) => { const resultat = a + b; return resultat * 2; };

Paramètres par défaut

function saluer(nom = 'Invité') { return `Bonjour ${nom}`; } saluer(); // 'Bonjour Invité' saluer('Alice'); // 'Bonjour Alice'

Rest Parameters & Spread

// Rest parameters (...) function somme(...nombres) { return nombres.reduce((acc, n) => acc + n, 0); } somme(1, 2, 3, 4); // 10 // Spread operator const arr1 = [1, 2, 3]; const arr2 = [...arr1, 4, 5]; // [1, 2, 3, 4, 5]

Callbacks & Higher Order Functions

// Fonction qui prend une fonction en paramètre function executerOperation(a, b, operation) { return operation(a, b); } const resultat = executerOperation(5, 3, (x, y) => x + y); console.log(resultat); // 8

🎨 Objets & POO

Création d'objets

// Objet littéral const personne = { nom: 'Alice', age: 30, saluer() { console.log(`Bonjour, je suis ${this.nom}`); } }; // Accès aux propriétés personne.nom; // 'Alice' personne['age']; // 30

Classes (ES6)

class Animal { constructor(nom) { this.nom = nom; } parler() { console.log(`${this.nom} fait du bruit`); } } class Chien extends Animal { constructor(nom, race) { super(nom); this.race = race; } parler() { console.log(`${this.nom} aboie`); } } const chien = new Chien('Rex', 'Labrador'); chien.parler(); // 'Rex aboie'

Méthodes d'objets

const obj = { a: 1, b: 2, c: 3 }; // Object.keys() - retourne les clés Object.keys(obj); // ['a', 'b', 'c'] // Object.values() - retourne les valeurs Object.values(obj); // [1, 2, 3] // Object.entries() - retourne clés et valeurs Object.entries(obj); // [['a', 1], ['b', 2], ['c', 3]] // Object.assign() - copie/fusion d'objets const copie = Object.assign({}, obj);

Spread avec objets

const obj1 = { a: 1, b: 2 }; const obj2 = { c: 3, d: 4 }; // Fusion d'objets const fusion = { ...obj1, ...obj2 }; // { a: 1, b: 2, c: 3, d: 4 } // Copie avec modification const modifie = { ...obj1, b: 10 }; // { a: 1, b: 10 }

📊 Tableaux (Arrays)

Méthodes de base

const fruits = ['pomme', 'banane', 'orange']; // Ajouter/retirer fruits.push('kiwi'); // Ajoute à la fin fruits.pop(); // Retire le dernier fruits.unshift('fraise'); // Ajoute au début fruits.shift(); // Retire le premier // Autres méthodes fruits.length; // Nombre d'éléments fruits.includes('pomme'); // true fruits.indexOf('banane'); // 1

map(), filter(), reduce()

const nombres = [1, 2, 3, 4, 5]; // map() - transforme chaque élément const doubles = nombres.map(n => n * 2); // [2, 4, 6, 8, 10] // filter() - filtre les éléments const pairs = nombres.filter(n => n % 2 === 0); // [2, 4] // reduce() - réduit à une seule valeur const somme = nombres.reduce((acc, n) => acc + n, 0); // 15

find(), findIndex(), some(), every()

const users = [ { id: 1, nom: 'Alice', age: 25 }, { id: 2, nom: 'Bob', age: 30 }, { id: 3, nom: 'Charlie', age: 35 } ]; // find() - trouve le premier élément const user = users.find(u => u.age > 28); // { id: 2, nom: 'Bob', age: 30 } // some() - au moins un élément correspond users.some(u => u.age > 30); // true // every() - tous les éléments correspondent users.every(u => u.age > 20); // true

sort() & reverse()

const nombres = [3, 1, 4, 1, 5, 9]; // Tri croissant nombres.sort((a, b) => a - b); // Tri décroissant nombres.sort((a, b) => b - a); // Inverser le tableau nombres.reverse();

⏱️ Programmation Asynchrone

Callbacks

// Fonction avec callback function chargerDonnees(callback) { setTimeout(() => { const donnees = { id: 1, nom: 'Alice' }; callback(donnees); }, 1000); } chargerDonnees((data) => { console.log(data); });

Promises

// Créer une Promise const maPromise = new Promise((resolve, reject) => { const succes = true; if (succes) { resolve('Succès!'); } else { reject('Erreur!'); } }); // Utiliser une Promise maPromise .then(resultat => console.log(resultat)) .catch(erreur => console.error(erreur)) .finally(() => console.log('Terminé'));

async / await

// Fonction asynchrone async function chargerDonnees() { try { const response = await fetch('https://api.example.com/data'); const data = await response.json(); return data; } catch (erreur) { console.error('Erreur:', erreur); } } // Utilisation chargerDonnees().then(data => console.log(data));

Fetch API

// GET request fetch('https://api.example.com/users') .then(response => response.json()) .then(data => console.log(data)) .catch(error => console.error(error)); // POST request fetch('https://api.example.com/users', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ nom: 'Alice', age: 25 }) }) .then(response => response.json()) .then(data => console.log(data));

✨ Fonctionnalités ES6+

Sets & Maps

// Set - collection de valeurs uniques const monSet = new Set([1, 2, 3, 2, 1]); console.log(monSet); // Set { 1, 2, 3 } monSet.add(4); monSet.has(2); // true monSet.delete(1); // Map - collection clé-valeur const maMap = new Map(); maMap.set('nom', 'Alice'); maMap.set('age', 30); maMap.get('nom'); // 'Alice' maMap.has('age'); // true

Symbols

// Créer un Symbol unique const id = Symbol('id'); const id2 = Symbol('id'); console.log(id === id2); // false // Utilisation comme clé d'objet const user = { nom: 'Alice', [id]: 123 };

Generators

// Fonction génératrice function* compter() { yield 1; yield 2; yield 3; } const compteur = compter(); console.log(compteur.next().value); // 1 console.log(compteur.next().value); // 2 console.log(compteur.next().value); // 3

Proxy

// Intercepter les opérations sur un objet const handler = { get(target, prop) { console.log(`Accès à ${prop}`); return target[prop]; } }; const obj = { nom: 'Alice' }; const proxy = new Proxy(obj, handler); proxy.nom; // Log: "Accès à nom", retourne "Alice"
